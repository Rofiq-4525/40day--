function computeProgress(){ const maxWeek = getMaxStoredWeek(); let attempts = 1, streak = 0, best = 0, completions = 0; /* সর্বশেষ দিন খুঁজে বের করা (যেদিন কোনো ডেটা আছে) */ let lastWeek = -1, lastDay = -1; for(let w = maxWeek; w >= 1; w--){ const data = loadWeekData(w); for(let d = dayNames.length - 1; d >= 0; d--){ const states = prayers.map(p => data[d + "_" + p.key] || ""); if(states.some(s => s !== "")){ lastWeek = w; lastDay = d; break; } } if(lastWeek !== -1) break; } for(let w = 1; w <= maxWeek; w++){ const data = loadWeekData(w); for(let d = 0; d < dayNames.length; d++){ const states = prayers.map(p => data[d + "_" + p.key] || ""); const isLastDay = (w === lastWeek && d === lastDay); if(states.includes("cross")){ attempts += 1; streak = 0; } else if(states.every(s => s === "check")){ streak += 1; best = Math.max(best, streak); if(streak % 40 === 0){ completions += 1; } } else if(isLastDay && states.some(s => s === "check")){ /* * সর্বশেষ দিন: কিছু চেক আছে, কিন্তু সব নয় + কোনো ক্রস নেই। * এটাকেও ধারাবাহিকতায় যোগ করা হলো, যাতে লিডারবোর্ডে * সর্বশেষ দিনের অগ্রগতি সহ দেখায়। */ streak += 1; best = Math.max(best, streak); } } } return { streak, attempts, best, completions }; }
